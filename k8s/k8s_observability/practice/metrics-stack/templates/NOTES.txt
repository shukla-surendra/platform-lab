  password: {{ $kps.grafana.adminPassword }}

Prometheus (PromQL / target list):
  # "prometheus-operated" is a fixed name the Prometheus Operator itself creates
  # (not templated off the release name, unlike the chart's own
  # <release>-kube-promethe-prometheus Service, which gets truncated to fit the
  # 63-char DNS limit at a point that depends on release-name length) — always
  # correct regardless of release name, verified live against this release.
  kubectl -n {{ .Release.Namespace }} port-forward svc/prometheus-operated 9090:9090
  open http://localhost:9090/targets

Dashboards are already provisioned (defaultDashboardsEnabled: true) — open Grafana,
Dashboards, and look under the "Kubernetes" and "Node Exporter" folders. No manual
import needed.
